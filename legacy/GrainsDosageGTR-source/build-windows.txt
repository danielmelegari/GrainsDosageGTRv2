name: Build Windows VST3 and Installer
on:
  workflow_dispatch:
  push:
    branches: [main]
    paths: ['src/**', 'assets/**', 'CMakeLists.txt', 'test_*.cpp', 'installer/**', 'INSTALL-WINDOWS.txt', '.github/workflows/build-windows.yml']
permissions:
  contents: read
jobs:
  build:
    runs-on: windows-2022
    timeout-minutes: 30
    defaults:
      run:
        shell: pwsh
    steps:
      - uses: actions/checkout@v4
      - name: Locate sources and pinned SDK
        run: |
          $ErrorActionPreference = 'Stop'
          $PSNativeCommandUseErrorActionPreference = $true
          if (Test-Path 'GrainsDosage/CMakeLists.txt') { $project = Join-Path $env:GITHUB_WORKSPACE 'GrainsDosage' }
          elseif (Test-Path 'CMakeLists.txt') { $project = $env:GITHUB_WORKSPACE }
          else { throw 'Upload the extracted source files, not only the ZIP.' }
          "PROJECT_DIR=$project" >> $env:GITHUB_ENV
          $sdk = Join-Path $env:RUNNER_TEMP 'vst3sdk'
          git clone --no-checkout https://github.com/steinbergmedia/vst3sdk.git $sdk
          git -C $sdk checkout 3cdf9ca5d1f5b1b21e0a86832aa4abe55607bd96
          git -C $sdk submodule update --init --recursive --depth 1
      - name: Configure and build x64
        run: |
          $ErrorActionPreference = 'Stop'
          $PSNativeCommandUseErrorActionPreference = $true
          Set-Location $env:PROJECT_DIR
          cmake --fresh -S . -B build -G "Visual Studio 17 2022" -A x64 "-DVST3_SDK_ROOT=$env:RUNNER_TEMP/vst3sdk" -DSMTG_CREATE_PLUGIN_LINK=OFF -DGRAINS_BUILD_TESTS=ON -DSMTG_USE_STATIC_CRT=ON -DCMAKE_MSVC_RUNTIME_LIBRARY=MultiThreaded
          cmake --build build --config Release --parallel 4 2>&1 | Tee-Object "$env:GITHUB_WORKSPACE/build-windows.log"
      - name: Test DSP and native Windows GUI
        run: |
          $ErrorActionPreference = 'Stop'
          $PSNativeCommandUseErrorActionPreference = $true
          Set-Location $env:PROJECT_DIR
          ctest --test-dir build -C Release --output-on-failure
          & "build/gui-test/Release/aztec_gui_smoke_win.exe"
      - name: Verify and package VST3
        run: |
          $ErrorActionPreference = 'Stop'
          $PSNativeCommandUseErrorActionPreference = $true
          $plugin = Join-Path $env:PROJECT_DIR 'build/VST3/Release/GrainsDosage.vst3'
          $binary = Join-Path $plugin 'Contents/x86_64-win/GrainsDosage.vst3'
          if (!(Test-Path $binary)) { throw "Missing x64 plugin: $binary" }
          $bytes = [IO.File]::ReadAllBytes($binary)
          $pe = [BitConverter]::ToInt32($bytes, 0x3c)
          if ([BitConverter]::ToUInt16($bytes, $pe + 4) -ne 0x8664) { throw 'Plugin is not x64' }
          $dist = Join-Path $env:GITHUB_WORKSPACE 'dist'
          New-Item -ItemType Directory -Force $dist | Out-Null
          Compress-Archive -Path $plugin -DestinationPath "$dist/GrainsDosage-Windows10-x64.zip" -Force
      - name: Build Windows installer
        run: |
          $ErrorActionPreference = 'Stop'
          $PSNativeCommandUseErrorActionPreference = $true
          $cmake = Get-Content (Join-Path $env:PROJECT_DIR 'CMakeLists.txt') -Raw
          if ($cmake -notmatch 'project\(GrainsDosage VERSION ([0-9]+\.[0-9]+\.[0-9]+)') { throw 'Project version not found' }
          $version = $Matches[1]
          $iscc = Join-Path ${env:ProgramFiles(x86)} 'Inno Setup 6/ISCC.exe'
          if (!(Test-Path $iscc)) { $iscc = (Get-Command ISCC.exe -ErrorAction Stop).Source }
          $dist = Join-Path $env:GITHUB_WORKSPACE 'dist'
          & $iscc "/DAppVersion=$version" "/DProjectDir=$env:PROJECT_DIR" "/DOutputDir=$dist" (Join-Path $env:PROJECT_DIR 'installer/GrainsDosage.iss')
          $setup = Join-Path $dist "GrainsDosage-$version-Windows10-x64-Setup.exe"
          if (!(Test-Path $setup)) { throw 'Installer was not generated' }
          "SETUP_EXE=$setup" >> $env:GITHUB_ENV
          Copy-Item (Join-Path $env:PROJECT_DIR 'INSTALL-WINDOWS.txt') $dist
      - name: Test installation and uninstallation on the runner
        run: |
          $ErrorActionPreference = 'Stop'
          $installed = Join-Path $env:CommonProgramFiles 'VST3/GrainsDosage.vst3/Contents/x86_64-win/GrainsDosage.vst3'
          if (Test-Path $installed) { throw 'Unexpected existing installation on the clean runner' }
          $installLog = Join-Path $env:GITHUB_WORKSPACE 'installer-test.log'
          $result = Start-Process -FilePath $env:SETUP_EXE -ArgumentList @('/VERYSILENT', '/SUPPRESSMSGBOXES', '/NORESTART', "/LOG=`"$installLog`"") -Wait -PassThru
          if ($result.ExitCode -ne 0) { throw "Installer failed: $($result.ExitCode)" }
          if (!(Test-Path $installed)) { throw 'Installed plugin missing' }
          $source = Join-Path $env:PROJECT_DIR 'build/VST3/Release/GrainsDosage.vst3/Contents/x86_64-win/GrainsDosage.vst3'
          if ((Get-FileHash $installed).Hash -ne (Get-FileHash $source).Hash) { throw 'Installed binary differs from tested binary' }
          $uninstaller = Join-Path $env:ProgramFiles 'GrainsDosage/unins000.exe'
          $result = Start-Process -FilePath $uninstaller -ArgumentList @('/VERYSILENT', '/SUPPRESSMSGBOXES', '/NORESTART') -Wait -PassThru
          if ($result.ExitCode -ne 0 -or (Test-Path $installed)) { throw 'Uninstall verification failed' }
      - name: Generate checksums
        run: |
          $ErrorActionPreference = 'Stop'
          Get-ChildItem "$env:GITHUB_WORKSPACE/dist" -File | Where-Object { $_.Extension -in '.exe','.zip' } | ForEach-Object { "$((Get-FileHash $_.FullName -Algorithm SHA256).Hash.ToLower())  $($_.Name)" } | Set-Content "$env:GITHUB_WORKSPACE/dist/SHA256SUMS.txt"
      - uses: actions/upload-artifact@v4
        with:
          name: GrainsDosage-Windows10-x64-Installer
          path: dist/
          if-no-files-found: error
          retention-days: 14
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: Windows-build-diagnostics
          path: |
            build-windows.log
            installer-test.log
            **/Testing/Temporary/LastTest.log
          if-no-files-found: ignore
          retention-days: 7
